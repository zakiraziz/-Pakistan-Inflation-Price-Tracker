  71| def asset_version():
  72|     """Cache-busting build id: newest mtime of the front-end assets.
  73| 
  74|     app.js / helpers.js / live.js / style.css are served with a ?v=<id> stamp so
  75|     a browser can never keep running an old bundle against new markup (which
  76|     silently breaks the dashboard: listeners wired but nothing renders).
  77|     """
  78|     newest = 0.0
  79|     for name in ("app.js", "helpers.js", "live.js", "style.css"):
  80|         with contextlib.suppress(OSError):  # pragma: no cover - asset always shipped
  81|             newest = max(newest, os.path.getmtime(os.path.join(BASE_DIR, "static", name)))
  82|     return str(int(newest))

 167| def _series_between(start, end, ids, status=db.STATUS_APPROVED):
 168|     con = db.connect()
 169|     if ids is None:
 170|         ids = [r["id"] for r in db.get_items(con)]
 171|     ph = ",".join("?" * len(ids))
 172|     q = (
 173|         "SELECT i.name AS name, p.date AS date, p.price AS price "
 174|         "FROM prices p JOIN items i ON i.id = p.item_id "
 175|         f"WHERE p.status = ? AND p.item_id IN ({ph})"
 176|     )
 177|     params = [status, *ids]
 178|     if start:
 179|         q += " AND p.date >= ?"
 180|         params.append(start)
 181|     if end:
 182|         q += " AND p.date <= ?"
 183|         params.append(end)
 184|     q += " ORDER BY p.date"
 185|     rows = con.execute(q, params).fetchall()
 186|     con.close()
 187|     return rows

 214| @app.route("/api/metrics")
 215| @cache.cached(query_string=True)
 216| def api_metrics():
 217|     """Summary stats for the selected window (drives the dashboard cards)."""
 218|     start, end, ids = _parse_window()
 219|     rows = _series_between(start, end, ids)
 220|     if not rows:
 221|         return jsonify({"count": 0, "weeks": 0})
 222|     by = {}
 223|     for r in rows:
 224|         by.setdefault(r["name"], []).append(r["price"])
 225|     names = list(by)
 226|     first_tot = sum(v[0] for v in by.values())
 227|     last_tot = sum(v[-1] for v in by.values())
 228|     basket_pct = (last_tot - first_tot) / first_tot * 100 if first_tot else 0.0
 229| 
 230|     stats = []
 231|     for name, pxs in by.items():
 232|         if len(pxs) >= 2:
 233|             stats.append((name, (pxs[-1] - pxs[0]) / pxs[0] * 100))
 234|     riser = max(stats, key=lambda t: t[1]) if stats else None
 235|     faller = min(stats, key=lambda t: t[1]) if stats else None
 236|     dates = sorted({r["date"] for r in rows})
 237|     return jsonify(
 238|         {
 239|             "count": len(names),
 240|             "weeks": len(dates),
 241|             "first_date": dates[0],
 242|             "last_date": dates[-1],
 243|             "basket_pct": round(basket_pct, 2),
 244|             "start_total": round(first_tot, 2),
 245|             "end_total": round(last_tot, 2),
 246|             "biggest_riser": {"name": riser[0], "pct": round(riser[1], 2)} if riser else None,
 247|             "biggest_faller": {"name": faller[0], "pct": round(faller[1], 2)} if faller else None,
 248|         }

 256|     start, end, ids = _parse_window()
 257|     con = db.connect()
 258|     if ids is None:
 259|         ids = [r["id"] for r in db.get_items(con)]
 260|     ph = ",".join("?" * len(ids))
 261|     q = (
 262|         "SELECT i.name AS name, i.category AS category, i.unit AS unit, "
 263|         "       p.date AS date, p.price AS price "
 264|         "FROM prices p JOIN items i ON i.id = p.item_id "

 268|     if start:
 269|         q += " AND p.date >= ?"

 272|         q += " AND p.date <= ?"
 273|         params.append(end)
 274|     q += " ORDER BY p.date"
 275|     rows = con.execute(q, params).fetchall()
 276|     con.close()
 277| 
 278|     dates, seen, by = [], {}, {}
 279|     for r in rows:
 280|         if r["date"] not in seen:
 281|             seen[r["date"]] = 1
 282|             dates.append(r["date"])
 283|         by.setdefault(r["name"], {"category": r["category"], "unit": r["unit"], "prices": []})[
 284|             "prices"
 285|         ].append(r["price"])
 286| 
 287|     items_out = []
 288|     for name, d in by.items():
 289|         px = d["prices"]
 290|         first, last = px[0], px[-1]
 291|         items_out.append(
 292|             {
 293|                 "name": name,
 294|                 "category": d["category"],
 295|                 "unit": d["unit"],
 296|                 "first": first,
 297|                 "last": last,
 298|                 "pct": round((last - first) / first * 100, 2) if first else 0.0,
 299|                 "prices": px,
 300|             }
 301|         )
 302| 
 303|     index = []
 304|     if dates and by:
 305|         total0 = sum(items_out[i]["prices"][0] for i in range(len(items_out)))
 306|         for i in range(len(dates)):
 307|             total = sum(items_out[j]["prices"][i] for j in range(len(items_out)))
 308|             index.append(round(total / total0 * 100, 2))
 309| 
 310|     return jsonify({"dates": dates, "index": index, "items": items_out})

 313| @app.route("/api/inflation")
 314| @cache.cached(query_string=True)
 315| def api_inflation():
 316|     """Headline inflation stats for the window (annualised, weekly, YoY)."""
 317|     start, end, ids = _parse_window()
 318|     rows = _series_between(start, end, ids)
 319|     if not rows:
 320|         return jsonify({"weeks": 0})
 321|     sums = {}
 322|     for r in rows:
 323|         sums[r["date"]] = sums.get(r["date"], 0.0) + r["price"]
 324|     dates = sorted(sums)
 325|     base = sums[dates[0]]
 326|     last_tot = sums[dates[-1]]
 327|     basket_pct = (last_tot - base) / base * 100
 328|     days = (len(dates) - 1) * 7
 329|     annualized = ((last_tot / base) ** (365.0 / days) - 1) * 100 if days > 0 else 0.0
 330|     avg_weekly = ((last_tot / base) ** (1.0 / max(1, len(dates) - 1)) - 1) * 100
 331| 
 332|     from datetime import date as _date
 333|     from datetime import timedelta
 334| 
 335|     last_d = _date.fromisoformat(dates[-1])
 336|     target = last_d - timedelta(days=364)
 337|     yoy = None
 338|     for dt in dates:
 339|         if _date.fromisoformat(dt) >= target:
 340|             yoy = (last_tot / sums[dt] - 1) * 100
 341|             break
 342|     return jsonify(
 343|         {
 344|             "weeks": len(dates),
 345|             "first_date": dates[0],
 346|             "last_date": dates[-1],
 347|             "basket_pct": round(basket_pct, 2),
 348|             "annualized_pct": round(annualized, 2),
 349|             "avg_weekly_pct": round(avg_weekly, 2),
 350|             "yoy_pct": round(yoy, 2) if yoy is not None else None,
 351|         }
 352|     )

 410| @app.route("/api/stream")
 411| @limiter.exempt
 412| def api_stream():
 413|     """Server-Sent Events feed that keeps every open dashboard current.
 414| 
 415|     Emits ``hello`` on connect, ``update`` the instant the dataset changes
 416|     (ingest, approval, or a write by another process), and ``heartbeat`` every
 417|     ``live.HEARTBEAT_SECONDS`` so a client can tell a quiet week from a dead
 418|     connection. The stream is long-lived, so it is exempt from the request rate
 419|     limit - one browser holds exactly one connection.
 420|     """
 421|     live.start_watcher()
 422| 
 423|     def frames():
 424|         yield "retry: 3000\n\n"
 425|         yield live.sse_frame("hello", live.snapshot())
 426|         seen = live.tick()
 427|         while True:
 428|             current = live.wait(seen, live.HEARTBEAT_SECONDS)
 429|             changed = current != seen
 430|             seen = current
 431|             yield live.sse_frame("update" if changed else "heartbeat", live.snapshot())
 432| 
 433|     resp = Response(frames(), mimetype="text/event-stream")
 434|     resp.headers["Cache-Control"] = "no-store"
 435|     resp.headers["X-Accel-Buffering"] = "no"  # never buffer behind a proxy
 436|     resp.headers["Connection"] = "keep-alive"
 437|     return resp
